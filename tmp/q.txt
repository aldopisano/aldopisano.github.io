function creaTuttiIQuestionari() {
  crea_(PARAGRAFI);
}

function creaUnSoloQuestionario() {
  const CODICE = '1.1'; // scrivi qui il paragrafo da rifare
  crea_(PARAGRAFI.filter(function (p) { return p.codice === CODICE; }));
}

function crea_(elenco) {
  if (!elenco.length) throw new Error('Nessun paragrafo con quel codice.');
  const cartella = cartellaQuestionari_();
  const ss = foglioRisposte_(cartella);

  const moduli = [];
  elenco.forEach(function (p) {
    const form = creaModulo_(p);
    sposta_(form.getId(), cartella);
    collegaAlFoglio_(form, ss.getId(), 'Risposte ' + p.codice);
    let link = form.getPublishedUrl();
    try { link = form.shortenFormUrl(link); } catch (e) { /* resta il link lungo */ }
    moduli.push(['§ ' + p.codice, p.titolo, link, form.getEditUrl()]);
    Logger.log('§ ' + p.codice + ' – link per le classi: ' + link);
  });

  scriviSchedeDiServizio_(ss.getId(), moduli);
  const doc = documentoTesti_(cartella);
  Logger.log('Cartella «' + NOME_CARTELLA + '»: ' + cartella.getUrl());
  Logger.log('Foglio delle risposte: ' + ss.getUrl());
  Logger.log('Documento dei testi: ' + doc.getUrl());
}

function creaModulo_(p) {
  const form = FormApp.create(TITOLO_MODULO + p.codice);
  impostazioni_(form);
  form.setDescription(informativa_());

  // Logo AI & Ethics Literacy in apertura (preso dal sito del libro)
  const logo = logo_();
  if (logo) form.addImageItem().setImage(logo).setTitle('AI & Ethics Literacy for Students')
    .setAlignment(FormApp.Alignment.CENTER).setWidth(320);

  // 1. Informativa e consenso (nella descrizione del modulo)
  const consenso = form.addMultipleChoiceItem().setTitle(CONSENSO.titolo).setRequired(true);

  // 2. La scuola
  form.addPageBreakItem().setTitle('La tua scuola')
    .setHelpText('È l’unica informazione su di te che ti chiediamo.');
  sceltaSingola_(form, 'Che tipo di scuola frequenti?', TIPI_SCUOLA);

  // 3. Riassunto delle parti precedenti e consegna di lettura
  form.addPageBreakItem().setTitle(p.riquadro).setHelpText(testoRiassunto_(p));

  // 4. La lettura
  pagina_(form, PAGINE.lettura, false);
  PAGINE.lettura.domande.forEach(function (d) { sceltaSingola_(form, d.testo, d.opzioni); });

  // 5-6. Scrittura e contenuti
  paginaScala_(form, PAGINE.scrittura, p);
  paginaScala_(form, PAGINE.contenuti, p);

  // 7. Inizio del capitolo (solo primo paragrafo di ogni capitolo)
  if (p.posizione === 1) paginaScala_(form, PAGINE.apertura, p);

  // 8. Comprensione, con una domanda sul contenuto
  paginaScala_(form, PAGINE.comprensione, p);
  sceltaSingola_(form, p.verifica.domanda, p.verifica.opzioni.concat(['Non lo so']));

  // 9. Fine del capitolo (solo ultimo paragrafo di ogni capitolo)
  if (p.posizione === 3) paginaScala_(form, PAGINE.chiusura, p);

  // 10. Accessibilità e domanda sul sito
  paginaScala_(form, PAGINE.accessibilita, p);
  const filtro = form.addMultipleChoiceItem().setTitle(PAGINE.accessibilita.filtro.testo).setRequired(true);

  // 11. Il sito (solo per chi l'ha usato)
  const paginaSito = pagina_(form, PAGINE.sito, true);
  caselle_(form, PAGINE.sito.dispositivo.testo, PAGINE.sito.dispositivo.opzioni, true);
  caselle_(form, PAGINE.sito.strumenti.testo, PAGINE.sito.strumenti.opzioni, true);
  PAGINE.sito.item.forEach(function (i) { scala_(form, testo_(i.testo, p), i.obbligatoria); });

  // 12. Per finire
  const paginaFinale = pagina_(form, PAGINE.finale, false);
  const v = PAGINE.finale.voto;
  form.addScaleItem().setTitle(v.testo).setBounds(v.min, v.max)
    .setLabels(v.etichettaMin, v.etichettaMax).setRequired(true);
  PAGINE.finale.aperte.forEach(function (a) {
    form.addParagraphTextItem().setTitle(a.testo)
      .setHelpText('Facoltativa. Non scrivere nomi né informazioni che permettano di riconoscere te o altre persone.')
      .setRequired(false);
  });

  // Percorsi
  consenso.setChoices([
    consenso.createChoice(CONSENSO.si, FormApp.PageNavigationType.CONTINUE),
    consenso.createChoice(CONSENSO.no, FormApp.PageNavigationType.SUBMIT)
  ]);
  const opzioniFiltro = PAGINE.accessibilita.filtro.opzioni;
  filtro.setChoices(opzioniFiltro.map(function (o, i) {
    return filtro.createChoice(o, i === 0 ? paginaSito : paginaFinale);
  }));

  pubblica_(form);
  return form;
}

let LOGO_CACHE_ = null;
function logo_() {
  if (LOGO_CACHE_) return LOGO_CACHE_.copyBlob();
  try {
    LOGO_CACHE_ = UrlFetchApp.fetch(URL_LOGO).getBlob().setName('logo-ael.jpg');
    return LOGO_CACHE_.copyBlob();
  } catch (e) {
    Logger.log('Logo non scaricato (' + e + '): il modulo viene creato senza.');
    return null;
  }
}

function impostazioni_(form) {
  try {
    form.setEmailCollectionType(FormApp.EmailCollectionType.DO_NOT_COLLECT);
  } catch (e) {
    form.setCollectEmail(false);
  }
  try { form.setRequireLogin(false); } catch (e) { /* opzione presente solo negli account scolastici o aziendali */ }
  form.setLimitOneResponsePerUser(false);
  form.setAllowResponseEdits(false);
  form.setShowLinkToRespondAgain(false);
  form.setPublishingSummary(false);
  form.setProgressBar(true);
  form.setShuffleQuestions(false);
  form.setConfirmationMessage(CONFERMA);
}

function pubblica_(form) {
  // Dal 2026 Google crea i moduli da script come «non pubblicati»: senza questo passo
  // il link non accetta risposte.
  try { form.setPublished(true); } catch (e) { /* versioni precedenti: il modulo è già pubblicato */ }
  form.setAcceptingResponses(true);
}

function informativa_() {
  const parere = PARERE_COMITATO ? ' (parere ' + PARERE_COMITATO + ')' : '';
  return INFORMATIVA.map(function (t) { return t.replace('{PARERE}', parere); }).join('\n\n');
}

function testoRiassunto_(p) {
  const righe = p.riassunto.map(function (r) { return '• ' + r[0] + ' – ' + r[1]; });
  let t = righe.join('\n\n') + '\n\n' + p.leggi;
  if (p.link) t += '\n\nPuoi leggere il paragrafo anche sul sito: ' + p.link;
  return t + '\n\nQuando hai finito di leggere, torna qui e premi «Avanti».';
}

function testo_(t, p) {
  return t.replace('{K2}', p.K2).replace('{BOX}', p.riquadro).replace('{A1}', p.A1 || '');
}

function pagina_(form, def, conScala) {
  const aiuto = conScala ? def.aiuto + '\n\n' + SCALA.legenda : def.aiuto;
  return form.addPageBreakItem().setTitle(def.titolo).setHelpText(aiuto);
}

function paginaScala_(form, def, p) {
  const pagina = pagina_(form, def, true);
  def.item.forEach(function (i) { scala_(form, testo_(i.testo, p), i.obbligatoria); });
  return pagina;
}

function scala_(form, titolo, obbligatoria) {
  return form.addScaleItem()
    .setTitle(titolo)
    .setBounds(SCALA.min, SCALA.max)
    .setLabels(SCALA.etichettaMin, SCALA.etichettaMax)
    .setRequired(obbligatoria);
}

function sceltaSingola_(form, titolo, opzioni) {
  const item = form.addMultipleChoiceItem().setTitle(titolo).setRequired(true);
  item.setChoices(opzioni.map(function (o) { return item.createChoice(o); }));
  return item;
}

function caselle_(form, titolo, opzioni, obbligatoria) {
  const item = form.addCheckboxItem().setTitle(titolo).setRequired(obbligatoria);
  item.setChoices(opzioni.map(function (o) { return item.createChoice(o); }));
  return item;
}

function cartellaQuestionari_() {
  const testo = String(CARTELLA_PROGETTO).trim();
  const m = testo.match(/folders\/([-\w]+)/) || testo.match(/^([-\w]{20,})$/);
  if (!m) {
    throw new Error('Incolla in CARTELLA_PROGETTO il link della cartella «AI and Ethics Literacy for Students».');
  }
  const madre = DriveApp.getFolderById(m[1]);
  const esistenti = madre.getFoldersByName(NOME_CARTELLA);
  return esistenti.hasNext() ? esistenti.next() : madre.createFolder(NOME_CARTELLA);
}

function foglioRisposte_(cartella) {
  const esistenti = cartella.getFilesByName(NOME_FOGLIO);
  if (esistenti.hasNext()) return SpreadsheetApp.openById(esistenti.next().getId());
  const ss = SpreadsheetApp.create(NOME_FOGLIO);
  sposta_(ss.getId(), cartella);
  return ss;
}

function sposta_(id, cartella) {
  DriveApp.getFileById(id).moveTo(cartella);
}

function collegaAlFoglio_(form, idFoglio, nome) {
  const prima = SpreadsheetApp.openById(idFoglio).getSheets().map(function (s) { return s.getSheetId(); });
  form.setDestination(FormApp.DestinationType.SPREADSHEET, idFoglio);
  SpreadsheetApp.flush();
  Utilities.sleep(1500);
  try {
    const ss = SpreadsheetApp.openById(idFoglio);
    const nuove = ss.getSheets().filter(function (s) { return prima.indexOf(s.getSheetId()) === -1; });
    if (!nuove.length) return;
    let finale = nome, n = 2;
    while (ss.getSheetByName(finale)) finale = nome + ' (' + (n++) + ')';
    nuove[0].setName(finale);
  } catch (e) {
    Logger.log('Scheda «' + nome + '» non rinominata: ' + e);
  }
}

function scheda_(ss, nome, posizione) {
  let sh = ss.getSheetByName(nome);
  if (!sh) {
    const libere = ss.getSheets().filter(function (s) { return !s.getFormUrl() && /^(Foglio|Sheet)\s?\d+$/.test(s.getName()); });
    sh = libere.length ? libere[0].setName(nome) : ss.insertSheet(nome);
  }
  ss.setActiveSheet(sh);
  ss.moveActiveSheet(posizione);
  return sh;
}

function scriviSchedeDiServizio_(idFoglio, moduli) {
  const ss = SpreadsheetApp.openById(idFoglio);

  // «Moduli»: aggiorna solo le righe dei paragrafi appena creati
  const shModuli = scheda_(ss, 'Moduli', 1);
  const mappa = {};
  if (shModuli.getLastRow() > 1) {
    shModuli.getRange(2, 1, shModuli.getLastRow() - 1, 4).getValues()
      .forEach(function (r) { if (r[0]) mappa[r[0]] = r; });
  }
  moduli.forEach(function (r) { mappa[r[0]] = r; });
  const righe = Object.keys(mappa).sort().map(function (k) { return mappa[k]; });
  shModuli.clear();
  scriviTabella_(shModuli, [['Paragrafo', 'Titolo', 'Link per le classi', 'Link per modificare il modulo']].concat(righe));

  const legenda = scheda_(ss, 'Legenda', 2);
  legenda.clear();
  scriviTabella_(legenda, [['Codice', 'Dimensione', 'Testo', 'Tipo di risposta', 'Chi la vede']].concat(righeLegenda_()));

  const chiave = scheda_(ss, 'Chiave verifica', 3);
  chiave.clear();
  scriviTabella_(chiave, [['Paragrafo', 'Domanda sul contenuto', 'Risposta corretta']].concat(
    PARAGRAFI.map(function (p) { return ['§ ' + p.codice, p.verifica.domanda, p.verifica.opzioni[p.verifica.corretta]]; })
  ));
}

function documentoTesti_(cartella) {
  // Se il documento esiste già non lo riscrive, per non perdere i commenti del gruppo di lavoro.
  const esistenti = cartella.getFilesByName(NOME_DOCUMENTO);
  if (esistenti.hasNext()) return esistenti.next();

  const doc = DocumentApp.create(NOME_DOCUMENTO);
  sposta_(doc.getId(), cartella);
  const body = doc.getBody();
  const H = DocumentApp.ParagraphHeading;
  function titolo(t, livello) { body.appendParagraph(t).setHeading(livello); }
  function testo(t) { body.appendParagraph(t); }
  function nota(t) { body.appendParagraph(t).editAsText().setItalic(true); }
  function voce(t) { body.appendListItem(t).setGlyphType(DocumentApp.GlyphType.BULLET); }

  body.getParagraphs()[0].setText('AI and Ethics Literacy for Students – Questionari di validazione').setHeading(H.TITLE);
  nota('Un questionario per ciascuno dei nove paragrafi. I riassunti sono ricavati dal testo dei tre capitoli.');

  titolo('Informativa e consenso', H.HEADING1);
  informativa_().split('\n\n').forEach(testo);
  testo(CONSENSO.titolo + ' (' + CONSENSO.si + ' / ' + CONSENSO.no + ')');

  titolo('Domande comuni a tutti i paragrafi', H.HEADING1);
  nota(SCALA.legenda);
  titolo('La tua scuola', H.HEADING2);
  voce('P1 – Che tipo di scuola frequenti? (' + TIPI_SCUOLA.join('; ') + ')');
  titolo(PAGINE.lettura.titolo, H.HEADING2);
  PAGINE.lettura.domande.forEach(function (d) { voce(d.codice + ' – ' + d.testo + ' (' + d.opzioni.join('; ') + ')'); });
  const generico = { K2: '[vedi il paragrafo]', riquadro: 'Finora nel libro', A1: '[vedi il paragrafo]' };
  [['scrittura', ''], ['contenuti', ''], ['apertura', ' – solo § 1.1, 2.1, 3.1'], ['comprensione', ''],
   ['chiusura', ' – solo § 1.3, 2.3, 3.3'], ['accessibilita', ''], ['sito', ' – solo chi ha usato il sito']]
    .forEach(function (c) {
      const def = PAGINE[c[0]];
      titolo(def.titolo + c[1], H.HEADING2);
      if (def.dispositivo) {
        voce(def.dispositivo.codice + ' – ' + def.dispositivo.testo + ' (' + def.dispositivo.opzioni.join('; ') + ')');
        voce(def.strumenti.codice + ' – ' + def.strumenti.testo + ' (' + def.strumenti.opzioni.join('; ') + ')');
      }
      def.item.forEach(function (i) { voce(i.codice + ' – ' + testo_(i.testo, generico) + (i.obbligatoria ? '' : ' (facoltativa)')); });
      if (c[0] === 'comprensione') voce('Q – Domanda sul contenuto, diversa per paragrafo (vedi sotto)');
      if (def.filtro) voce(def.filtro.codice + ' – ' + def.filtro.testo + ' (' + def.filtro.opzioni.join('; ') + ')');
    });
  titolo(PAGINE.finale.titolo, H.HEADING2);
  voce(PAGINE.finale.voto.codice + ' – ' + PAGINE.finale.voto.testo + ' (1-10)');
  PAGINE.finale.aperte.forEach(function (a) { voce(a.codice + ' – ' + a.testo + ' (facoltativa)'); });

  titolo('Paragrafo per paragrafo', H.HEADING1);
  PARAGRAFI.forEach(function (p) {
    titolo('§ ' + p.codice + ' – ' + p.titolo, H.HEADING2);
    titolo(p.riquadro, H.HEADING3);
    p.riassunto.forEach(function (r) { voce(r[0] + ' – ' + r[1]); });
    p.leggi.split('\n').forEach(testo);
    titolo('Item specifici', H.HEADING3);
    voce('K2 – ' + testo_(PAGINE.comprensione.item[1].testo, p));
    if (p.A1) voce('A1 – ' + p.A1);
    voce('Q – ' + p.verifica.domanda + ' (' + p.verifica.opzioni.map(function (o, i) {
      return 'ABCD'[i] + ') ' + o;
    }).join('; ') + '; Non lo so). Corretta: ' + 'ABCD'[p.verifica.corretta] + '.');
  });
  const id = doc.getId();
  doc.saveAndClose();
  return DriveApp.getFileById(id);
}

function righeLegenda_() {
  const scala = 'Scala ' + SCALA.min + '-' + SCALA.max;
  const righe = [['P0', 'Consenso', CONSENSO.titolo, 'Scelta singola', 'Tutti'],
                 ['P1', 'Tipo di scuola', 'Che tipo di scuola frequenti?', 'Scelta singola', 'Tutti']];
  PAGINE.lettura.domande.forEach(function (d) { righe.push([d.codice, d.dimensione, d.testo, 'Scelta singola', 'Tutti']); });
  const generico = { K2: '…', riquadro: '…', A1: '(frase sullo stimolo del capitolo)' };
  function aggiungi(def, chi) {
    def.item.forEach(function (i) {
      righe.push([i.codice, i.dimensione, testo_(i.testo, generico), scala + (i.obbligatoria ? '' : ' (facoltativa)'), chi]);
    });
  }
  aggiungi(PAGINE.scrittura, 'Tutti');
  aggiungi(PAGINE.contenuti, 'Tutti');
  aggiungi(PAGINE.apertura, 'Paragrafi 1.1, 2.1, 3.1');
  aggiungi(PAGINE.comprensione, 'Tutti');
  righe.push(['Q', 'Comprensione (verifica)', 'Domanda sul contenuto, diversa per paragrafo', 'Scelta singola con «Non lo so»', 'Tutti']);
  aggiungi(PAGINE.chiusura, 'Paragrafi 1.3, 2.3, 3.3');
  aggiungi(PAGINE.accessibilita, 'Tutti');
  const f = PAGINE.accessibilita.filtro;
  righe.push([f.codice, f.dimensione, f.testo, 'Scelta singola (filtro)', 'Tutti']);
  [PAGINE.sito.dispositivo, PAGINE.sito.strumenti].forEach(function (d) {
    righe.push([d.codice, d.dimensione, d.testo, 'Caselle', 'Chi ha usato il sito']);
  });
  aggiungi(PAGINE.sito, 'Chi ha usato il sito');
  const v = PAGINE.finale.voto;
  righe.push([v.codice, v.dimensione, v.testo, 'Scala ' + v.min + '-' + v.max, 'Tutti']);
  PAGINE.finale.aperte.forEach(function (a) { righe.push([a.codice, a.dimensione, a.testo, 'Testo libero (facoltativa)', 'Tutti']); });
  return righe;
}

function scriviTabella_(scheda, righe) {
  scheda.getRange(1, 1, righe.length, righe[0].length).setValues(righe);
  scheda.getRange(1, 1, 1, righe[0].length).setFontWeight('bold');
  scheda.setFrozenRows(1);
  scheda.autoResizeColumns(1, righe[0].length);
}
